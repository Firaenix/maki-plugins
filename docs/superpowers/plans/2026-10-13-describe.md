# describe plugin: implementation plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or
> superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** When a bash call needs approval, a panel over maki's permission prompt shows a one-sentence explanation the
agent wrote, plus the script formatted and highlighted. It offers a full-screen view, `y`/`n` answers, and a `Tab`
hand-off to the native prompt.

**Architecture:** A pure Lua plugin in this pack, with no maki core changes.
- Capture: a prompt hint asks the agent for an `intent` field. A `tool.bash.input` layer stores the call by tool id
  and strips that field.
- Decision logic: two pure modules, tested under the `luau` CLI.
  - `describe_store` remembers calls and tracks the open prompts per session.
  - `describe_view` turns a call into lines and keys into actions.
- `describe_panel` owns a focused `maki.ui` window, the `shfmt` job and the `maki.session.answer_permission` calls.

**Tech Stack:** Luau (maki's plugin runtime), the maki Lua API on fork/tip 0.6.3, `shfmt`, the `luau` CLI 0.738
from nixpkgs for unit tests, stylua 2.5, tmux for live checks.

**Spec:** `docs/superpowers/specs/2026-10-13-describe-design.md`. Read it first; this plan argues from it.

## Global Constraints

- **Version control**
  - jj only, never `git commit`/`checkout`/`branch`/`rebase`.
  - Each task starts its own change with `jj new -m "<message>"`.
  - Push nothing, in any repo.
- **Lua code**
  - Luau, formatted by stylua with this repo's `.stylua.toml`: `column_width = 100`, 2-space indent, double quotes.
  - No Luau-only syntax: no compound assignment and no type annotations. stylua must parse every file.
  - Comments explain why, not what, and wrap at word boundaries.
- **Never block a call:** the `tool.bash.input` layer always reaches `prev`, and every autocmd callback is wrapped so

- **Answers:** the plugin only ever answers `"allow"` or `"deny"` through `maki.session.answer_permission`. Rule-writing

- **Limits:** calls are capped at `MAX_CALLS = 64`, and the compact view shows `COMPACT_ROWS = 8` code rows.

## Review Focus

1. **A prompt in a background session** (several sessions in one TUI). The panel must not open and must not take the
   keys of the session being typed in. Pinned in Task 1.
2. **A prompt describe has nothing for** (another tool, a nested call) at the front of the native queue, with a
   captured bash prompt behind it. The panel stays closed until the front one resolves, so `y` never answers a call
   other than the one the native prompt shows. Pinned in Task 1.
3. **Raw escape bytes, tabs, carriage returns or invalid UTF-8 in a command.** They show as visible text and never
   reach the terminal. Pinned in Task 2.
4. **A single very long line**, such as a minified one-liner with no newlines. It wraps under one line number, and
   the compact view says the rest is in the full view. Pinned in Task 2.
5. **`shfmt` finishing after its prompt was answered.** The result is dropped and the panel does not come back.
   Pinned in Task 1.

```lua
-- A small harness for describe's pure modules, run under the luau CLI:
--
--   nix run nixpkgs#luau -- tests/describe_spec.lua
--
-- maki's own plugin specs run inside a maki build. These modules make no maki
-- calls, so plain luau is enough and the pack needs no build to be tested.

local M = {}

local failures = {}
local passed = 0

local function dump(v)
  if type(v) ~= "table" then
    return type(v) == "string" and string.format("%q", v) or tostring(v)
  end
  local keys = {}
  for k in pairs(v) do
    keys[#keys + 1] = k
  end
  table.sort(keys, function(a, b)
    return tostring(a) < tostring(b)
  end)
  local parts = {}
  for _, k in ipairs(keys) do
    parts[#parts + 1] = tostring(k) .. "=" .. dump(v[k])
  end
  return "{" .. table.concat(parts, ", ") .. "}"
end

function M.case(name, fn)
  local ok, err = pcall(fn)
  if ok then
    passed = passed + 1
  else
    failures[#failures + 1] = name .. ": " .. tostring(err)
  end
end

function M.eq(actual, expected, msg)
  if actual ~= expected then
    local text = string.format("%s\nexpected: %s\n  actual: %s", msg or "", dump(expected), dump(actual))
    error(text, 2)
  end
end

-- Compares tables by content, so a test can state a whole rendered line.
function M.same(actual, expected, msg)
  local a, e = dump(actual), dump(expected)
  if a ~= e then
    error(string.format("%s\nexpected: %s\n  actual: %s", msg or "", e, a), 2)
  end
end

function M.report()
  for _, failure in ipairs(failures) do
    print("FAIL " .. failure)
  end
  print(string.format("%d passed, %d failed", passed, #failures))
  if #failures > 0 then
    error(#failures .. " failing case(s)", 0)
  end
end

return M
```

```lua
-- describe's memory between a bash call reaching `tool.bash.input` and the
-- permission prompt it may raise. Pure, with no maki calls, so tests/ runs it
-- under plain luau.
--
-- Calls are kept by tool id, which `PermissionRequested` reports as `tool_id`.
-- Prompts are kept per session in the order the native prompt queues them,
-- including the ones describe has nothing to show for. The panel shows only
-- when the native prompt's front request is one of describe's own, so the two
-- never disagree about which call `y` answers.

local M = {}

M.MAX_CALLS = 64
M.MAX_TEXT_BYTES = 400
local ELLIPSIS = "…"

-- Backs off to a character boundary, so a capped sentence never ends in half
-- a character.
local function cap_bytes(s, max)
  if #s <= max then
    return s
  end
  local cut = max
  while cut > 0 do
    local b = s:byte(cut + 1)
    if b < 0x80 or b >= 0xC0 then
      break
    end
    cut = cut - 1
  end
  return s:sub(1, cut) .. ELLIPSIS
end

local function clean_text(value)
  if type(value) ~= "string" then
    return nil
  end
  local s = value:gsub("%s+", " "):match("^ ?(.-) ?$")
  if s == "" then
    return nil
  end
  return cap_bytes(s, M.MAX_TEXT_BYTES)
end

local Calls = {}
Calls.__index = Calls

function M.calls(cap)
  return setmetatable({ cap = cap or M.MAX_CALLS, by_id = {}, order = {} }, Calls)
end

-- Nested calls arrive with an empty id and a call with no command has nothing
-- to show, so both are skipped.
function Calls:remember(tool_id, input)
  if type(tool_id) ~= "string" or tool_id == "" or type(input) ~= "table" then
    return
  end
  local command = input.command
  if type(command) ~= "string" or command == "" then
    return
  end
  if self.by_id[tool_id] == nil then
    self.order[#self.order + 1] = tool_id
  end
  self.by_id[tool_id] = {
    command = command,
    sentence = clean_text(input.intent) or clean_text(input.description),
    workdir = type(input.workdir) == "string" and input.workdir or nil,
  }
  while #self.order > self.cap do
    self.by_id[table.remove(self.order, 1)] = nil
  end
end

function Calls:get(tool_id)
  if tool_id == nil then
    return nil
  end
  return self.by_id[tool_id]
end

function Calls:forget(tool_id)
  if tool_id == nil or self.by_id[tool_id] == nil then
    return
  end
  self.by_id[tool_id] = nil
  for i, id in ipairs(self.order) do
    if id == tool_id then
      table.remove(self.order, i)
      return
    end
  end
end

local Prompts = {}
Prompts.__index = Prompts

function M.prompts()
  return setmetatable({ by_session = {}, focused = nil }, Prompts)
end

-- {call} is nil for a prompt describe has nothing to show for. It still takes
-- its place in the queue.
function Prompts:open(session_id, request_id, call)
  if session_id == nil or request_id == nil then
    return
  end
  local queue = self.by_session[session_id]
  if not queue then
    queue = {}
    self.by_session[session_id] = queue
  end
  queue[#queue + 1] = {
    request_id = request_id,
    session_id = session_id,
    call = call,
    handed_off = false,
  }
end

local function find(self, request_id)
  for _, queue in pairs(self.by_session) do
    for i, item in ipairs(queue) do
      if item.request_id == request_id then
        return queue, i, item
      end
    end
  end
  return nil
end

function Prompts:resolve(request_id)
  local queue, i = find(self, request_id)
  if queue then
    table.remove(queue, i)
  end
end

function Prompts:hand_off(request_id)
  local _, _, item = find(self, request_id)
  if item then
    item.handed_off = true
  end
end

-- False for a request no longer open: a formatter that finishes after its
-- prompt closed must not bring the panel back.
function Prompts:set_formatted(request_id, text)
  local _, _, item = find(self, request_id)
  if not item then
    return false
  end
  item.formatted = text
  return true
end

function Prompts:focus(session_id)
  self.focused = session_id
end

-- The front of the focused session's queue, when describe captured that call
-- and the user has not handed it off.
function Prompts:current()
  local queue = self.focused and self.by_session[self.focused]
  local front = queue and queue[1]
  if front and front.call and not front.handed_off then
    return front
  end
  return nil
end

return M
```

```lua
-- What the describe panel draws, and what its keys do. Pure, with no maki
-- calls: the panel passes in sizes, a display-width function and highlighted
-- rows, so tests/ runs this under plain luau.
--
-- Lines are in `buf:line` form: a string, or a list of `{ text, style }` spans.

local M = {}

M.COMPACT_ROWS = 8
-- open_win sizes the outer box, so the border takes a row at each end.
M.CHROME = 2
-- Rows the compact panel leaves free for the native prompt, whose tool line
-- and key hints must stay visible below it.
M.PROMPT_ROWS = 14
M.MIN_HEIGHT = 5

local GAP = "  "
local DIM = "dim"

local SCROLL = { ["j"] = 1, ["<Down>"] = 1, ["k"] = -1, ["<Up>"] = -1 }

local function plain_width(s)
  return utf8.len(s) or #s
end

-- A command can hold tabs, carriage returns and raw escape bytes. Drawn as
-- they are they would shift columns or reach the terminal, so they become
-- visible text.
function M.sanitize(text)
  text = text:gsub("\r\n", "\n"):gsub("\t", "  ")
  text = text:gsub("%c", function(c)
    if c == "\n" then
      return c
    end
    return string.format("\\x%02x", c:byte())
  end)
  if not utf8.len(text) then
    text = text:gsub("[\128-\255]", "?")
  end
  return text
end

-- Cuts {s} into pieces no wider than {width}, between characters. Code wraps
-- this way: breaking at spaces would move tokens to where they are not.
local function hard_wrap(s, width, dw)
  local out, cur, cur_w = {}, {}, 0
  for _, cp in utf8.codes(s) do
    local ch = utf8.char(cp)
    local w = dw(ch)
    if cur_w > 0 and cur_w + w > width then
      out[#out + 1] = table.concat(cur)
      cur, cur_w = {}, 0
    end
    cur[#cur + 1] = ch
    cur_w = cur_w + w
  end
  out[#out + 1] = table.concat(cur)
  return out
end

function M.wrap_words(text, width, dw)
  dw = dw or plain_width
  local lines, cur, cur_w = {}, nil, 0
  for word in text:gmatch("%S+") do
    local w = dw(word)
    if cur and cur_w + 1 + w <= width then
      cur, cur_w = cur .. " " .. word, cur_w + 1 + w
    else
      if cur then
        lines[#lines + 1] = cur
      end
      local pieces = w > width and hard_wrap(word, width, dw) or { word }
      for i = 1, #pieces - 1 do
        lines[#lines + 1] = pieces[i]
      end
      cur = pieces[#pieces]
      cur_w = dw(cur)
    end
  end
  if cur then
    lines[#lines + 1] = cur
  end
  return lines
end

-- One entry per screen row, with `lineno` only on the first row of each
-- source line, so a wrapped line still reads as one.
function M.rows(code, width, dw)
  dw = dw or plain_width
  local rows, n = {}, 0
  for line in (code .. "\n"):gmatch("(.-)\n") do
    n = n + 1
    for i, piece in ipairs(hard_wrap(line, width, dw)) do
      rows[#rows + 1] = { lineno = i == 1 and n or nil, text = piece }
    end
  end
  return rows
end

-- Everything the panel draws for one call at one width. {code} is the
-- sanitized command, formatted or as written; {inner} is the width inside the
-- border.
function M.content(call, code, inner, dw)
  dw = dw or plain_width
  code = code:gsub("\n+$", "")
  local total = select(2, code:gsub("\n", "")) + 1
  local gutter = #tostring(total) + #GAP
  local sentence = call.sentence and M.wrap_words(M.sanitize(call.sentence), inner, dw) or {}
  return {
    sentence = sentence,
    where = call.workdir and ("in " .. M.sanitize(call.workdir)) or nil,
    rows = M.rows(code, math.max(1, inner - gutter), dw),
    total = total,
    gutter = gutter,
  }
end

local function header(content)
  local out = {}
  for _, line in ipairs(content.sentence) do
    out[#out + 1] = line
  end
  if content.where then
    out[#out + 1] = { { content.where, DIM } }
  end
  if #out > 0 then
    out[#out + 1] = ""
  end
  return out
end

local function code_line(content, i)
  local row = content.rows[i]
  local num = row.lineno and tostring(row.lineno) or ""
  local pad = string.rep(" ", content.gutter - #GAP - #num)
  local line = { { pad .. num .. GAP, DIM } }
  local spans = content.styled and content.styled[i]
  if spans then
    for _, span in ipairs(spans) do
      line[#line + 1] = span
    end
  else
    line[#line + 1] = { row.text }
  end
  return line
end

function M.compact(content)
  local lines = header(content)
  local shown = math.min(#content.rows, M.COMPACT_ROWS)
  for i = 1, shown do
    lines[#lines + 1] = code_line(content, i)
  end
  if shown < #content.rows then
    local last = 0
    for i = shown, 1, -1 do
      if content.rows[i].lineno then
        last = content.rows[i].lineno
        break
      end
    end
    local more = content.total - last
    local note = "… v shows all"
    if more > 0 then
      note = string.format("… %d more lines, v shows all", more)
    end
    lines[#lines + 1] = { { note, DIM } }
  end
  return lines
end

function M.compact_height(line_count, term_rows)
  local room = term_rows - M.PROMPT_ROWS
  return math.max(M.MIN_HEIGHT, math.min(line_count + M.CHROME, room))
end

local function source(content)
  return content.formatted and "shfmt" or "as written"
end

function M.compact_title(content)
  return string.format(" bash · %d lines · %s ", content.total, source(content))
end

-- The header stays put in the full view and the code scrolls under it.
function M.visible_rows(content, height)
  return math.max(1, height - M.CHROME - #header(content))
end

function M.clamp_top(top, content, height)
  local max_top = math.max(1, #content.rows - M.visible_rows(content, height) + 1)
  return math.max(1, math.min(top, max_top))
end

function M.full(content, height, top)
  local lines = header(content)
  local last = math.min(#content.rows, top + M.visible_rows(content, height) - 1)
  for i = top, last do
    lines[#lines + 1] = code_line(content, i)
  end
  return lines
end

function M.full_title(content, top, height)
  local last = math.min(#content.rows, top + M.visible_rows(content, height) - 1)
  return string.format(" bash · %s · rows %d-%d of %d ", source(content), top, last, #content.rows)
end

function M.footer(state)
  local keys = {}
  if state.can_answer then
    keys[#keys + 1] = { "y", "allow" }
    keys[#keys + 1] = { "n", "deny" }
  end
  if state.mode == "full" then
    keys[#keys + 1] = { "j/k", "scroll" }
    keys[#keys + 1] = { "v", "back" }
  else
    keys[#keys + 1] = { "v", "full view" }
  end
  keys[#keys + 1] = { "Tab", "more answers" }
  return keys
end

-- "allow", "deny" or "hand_off" for the panel to act on, "redraw" after the
-- view changed, or nil for a key that does nothing here.
function M.handle_key(state, key, content, height)
  if key == "y" or key == "n" then
    if not state.can_answer then
      return nil
    end
    return key == "y" and "allow" or "deny"
  end
  if key == "<Tab>" or key == "<C-c>" then
    return "hand_off"
  end
  if state.mode == "compact" then
    if key == "v" then
      state.mode, state.top = "full", 1
      return "redraw"
    end
    return nil
  end
  if key == "v" or key == "<Esc>" then
    state.mode = "compact"
    return "redraw"
  end
  local page = M.visible_rows(content, height)
  local top = state.top
  if SCROLL[key] then
    top = top + SCROLL[key]
  elseif key == "<PageDown>" or key == "<Space>" then
    top = top + page
  elseif key == "<PageUp>" then
    top = top - page
  elseif key == "g" then
    top = 1
  elseif key == "G" then
    top = #content.rows
  else
    return nil
  end
  top = M.clamp_top(top, content, height)
  if top == state.top then
    return nil
  end
  state.top = top
  return "redraw"
end

return M
```

```lua
-- describe: say what a bash call does before you approve it.
--
-- The agent is asked for an `intent` sentence on any bash call beyond a single
-- simple command. A tool.bash.input layer keeps it, with the command, under
-- the call's id and strips it before bash parses the input. When the call
-- raises a permission prompt, a focused panel over the prompt shows the
-- sentence and the script, formatted by shfmt and highlighted. `y` and `n`
-- answer, `v` opens the whole script, and `Tab` hands the keys back to the
-- native prompt for its other answers.
--
-- Answering needs maki.session.answer_permission (fork branch
-- pr/answer-open-prompt) and the full trust this pack's plugin.toml grants.
-- Without it the panel still shows, and `Tab` is the way on.

local Store = require("describe_store")

local HINT = "- When a bash command is more than one simple command, also pass `intent`: one"
  .. " sentence of 20-30 words saying what the command does and why you are running it."
  .. " The user reads it when approving the call."

local calls = Store.calls()

-- A failure in here must never block a call or a prompt: it is logged, and
-- the native prompt carries on alone.
local function guard(name, fn)
  return function(...)
    local ok, err = pcall(fn, ...)
    if not ok then
      maki.log.warn("describe: " .. name .. ": " .. tostring(err))
    end
  end
end

local function data(ev)
  return ev.data or {}
end

local function without_intent(input)
  local copy = {}
  for k, v in pairs(input) do
    copy[k] = v
  end
  copy.intent = nil
  return copy
end

maki.api.register_prompt_hint({ slot = "tool_usage", content = HINT })

-- The layer runs before bash parses its input, so `intent` never reaches the
-- tool, and it always reaches `prev`, so nothing here can stop a call.
maki.api.set_slot("tool.bash.input", function(prev, input, ctx)
  guard("capture", function()
    calls:remember(ctx.tool_id, input)
  end)()
  if type(input) == "table" and input.intent ~= nil then
    return prev(without_intent(input), ctx)
  end
  return prev(input, ctx)
end)

maki.api.create_autocmd("ToolDone", {
  callback = guard("ToolDone", function(ev)
    calls:forget(data(ev).tool_id)
  end),
})
```

```lua
-- bash call the native prompt is showing. Layout and keys live in
-- describe_view; this owns the window, the shfmt job and the answers.
--
-- Focus is the point. While a permission prompt is open it takes every key
-- except those for a focused plugin window, so the panel has to hold focus to
-- hear `v`. Handing off closes it, which gives the keys back to the prompt.

local View = require("describe_view")

local M = {}
M.__index = M

-- Above plugin floats at the default zindex of 50.
local ZINDEX = 60
local SHFMT_TIMEOUT_MS = 1000
local SHFMT_TAIL = 1024
-- The command travels in the environment: jobstart has no stdin, and this way
-- nothing is written to disk or quoted for a shell.
local SHFMT = { "sh", "-c", 'printf %s "$DESCRIBE_CMD" | shfmt -i 2 -ci -bn' }

local function display_width(s)
  return maki.ui.display_width(s)
end

function M.new(prompts, can_answer)
  return setmetatable({
    prompts = prompts,
    can_answer = can_answer,
    win = nil,
    buf = nil,
    item = nil,
    state = nil,
    content = nil,
    -- Bumped on every build, so a build that finishes after a newer one
    -- started is thrown away.
    gen = 0,
  }, M)
end

-- Highlighting yields, so this runs in a task.
local function build(item)
  local inner = math.max(20, maki.ui.terminal_size().cols - View.CHROME)
  local code = View.sanitize(item.formatted or item.call.command)
  local content = View.content(item.call, code, inner, display_width)
  content.formatted = item.formatted ~= nil
  local texts = {}
  for i, row in ipairs(content.rows) do
    texts[i] = row.text
  end
  local ok, styled = pcall(maki.ui.highlight, table.concat(texts, "\n"), "bash")
  if ok and type(styled) == "table" and #styled == #content.rows then
    content.styled = styled
  end
  return content
end

function M:draw()
  local rows = maki.ui.terminal_size().rows
  local lines, config
  if self.state.mode == "full" then
    self.state.top = View.clamp_top(self.state.top, self.content, rows)
    lines = View.full(self.content, rows, self.state.top)
    config = { title = View.full_title(self.content, self.state.top, rows), height = "100%" }
  else
    lines = View.compact(self.content)
    config = {
      title = View.compact_title(self.content),
      height = View.compact_height(#lines, rows),
    }
  end
  config.footer = View.footer(self.state)
  self.buf:set_lines(lines)
  self.win:set_config(config)
end

-- A win is followed by PermissionResolved, which moves the panel on. A loss
-- means someone else answered first, and their PermissionResolved is already
-- on its way. Neither needs anything here.
function M:answer(item, decision)
  local ok, won, err =
    pcall(maki.session.answer_permission, item.request_id, { decision = decision })
  if ok and won ~= nil then
    return
  end
  maki.log.warn("describe: could not answer the prompt: " .. tostring(ok and err or won))
  self.can_answer = false
  if self.item == item then
    self.state.can_answer = false
    self:draw()
  end
end

function M:key(key)
  if not self.content then
    return
  end
  local rows = maki.ui.terminal_size().rows
  local action = View.handle_key(self.state, key, self.content, rows)
  if action == "redraw" then
    self:draw()
  elseif action == "hand_off" then
    self.prompts:hand_off(self.item.request_id)
    self:sync()
  elseif action == "allow" or action == "deny" then
    self:answer(self.item, action)
  end
end

function M:loop(win)
  while true do
    local ev = win:recv()
    if not ev or ev.type == "close" then
      if win == self.win then
        self.win, self.buf, self.item, self.content = nil, nil, nil, nil
      end
      return
    end
    -- Only the live window acts; one closed under a newer one just drains.
    if win == self.win and self.item then
      if ev.type == "key" then
        self:key(ev.key)
      elseif ev.type == "resize" then
        self:show(self.item)
      end
    end
  end
end

function M:open()
  self.buf = maki.ui.buf({ scratch = true })
  self.win = maki.ui.open_win(self.buf, {
    anchor = "NW",
    row = 0,
    col = 0,
    width = "100%",
    height = View.MIN_HEIGHT,
    border = "rounded",
    focus = true,
    zindex = ZINDEX,
  })
  local win = self.win
  maki.async.spawn(function()
    self:loop(win)
  end)
end

function M:close()
  local win = self.win
  self.win, self.buf, self.item, self.content = nil, nil, nil, nil
  if win then
    win:close()
  end
end

-- Builds and draws {item}. A build for a request the panel has since moved off
-- is thrown away.
function M:show(item)
  self.gen = self.gen + 1
  local gen = self.gen
  maki.async.spawn(function()
    local content = build(item)
    if gen ~= self.gen or self.item ~= item then
      return
    end
    self.content = content
    if not self.win then
      self:open()
    end
    self:draw()
  end)
end

-- Matches the panel to what the native prompt is showing. Called after every
-- change to the prompts.
function M:sync()
  local item = self.prompts:current()
  if item == self.item then
    return
  end
  if not item then
    self:close()
    return
  end
  self.item = item
  self.content = nil
  self.state = { mode = "compact", top = 1, can_answer = self.can_answer }
  self:show(item)
end

-- Redraws with the formatted script once shfmt is done, keeping the view and
-- scroll position the user is on.
function M:refresh(request_id)
  if self.item and self.item.request_id == request_id then
    self:show(self.item)
  end
end

function M:format(request_id, command)
  maki.async.spawn(function()
    local id = maki.fn.jobstart(SHFMT, { env = { DESCRIBE_CMD = command }, tail = SHFMT_TAIL })
    if not id then
      return
    end
    -- Straight after jobstart: a job waited on while it runs hands back every
    -- line, while one that already exited answers only from its tail.
    local res = maki.fn.jobwait(id, SHFMT_TIMEOUT_MS)
    if not res then
      pcall(maki.fn.jobstop, id)
      return
    end
    if res.exit_code ~= 0 or res.truncated or (res.stdout or "") == "" then
      return
    end
    if self.prompts:set_formatted(request_id, res.stdout) then
      self:refresh(request_id)
    end
  end)
end

return M
```
